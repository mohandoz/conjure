    # Use a while-read loop (not process substitution) to stay POSIX 3.2+ compliant
    REF_WITHOUT_SHA="$(jq -r '
      (.extraKnownMarketplaces // {}) | to_entries[] |
      select((.value.source.ref != null) and (.value.source.sha == null)) |
      .key' .claude/settings.json 2>/dev/null)" || true
    if [ -n "$REF_WITHOUT_SHA" ]; then
      # Iterate over newline-separated names (POSIX-safe: use IFS + read from heredoc)
      while IFS= read -r mkt_name; do
        [ -z "$mkt_name" ] && continue
        warn "[plugin] extraKnownMarketplaces '$mkt_name': has ref but no sha — re-run: conjure publish-plugin --marketplace"
      done <<MKT_NAMES_EOF
$REF_WITHOUT_SHA
MKT_NAMES_EOF
    else
      # Only emit ok if extraKnownMarketplaces has at least one entry
      EXTMKT_COUNT="$(jq '(.extraKnownMarketplaces // {}) | length' .claude/settings.json 2>/dev/null || echo 0)"
      if [ "$EXTMKT_COUNT" -gt 0 ]; then
        ok "all extraKnownMarketplaces entries have sha pinned"
      fi
    fi
  fi

POSIX 3.2+ compliance: no process substitution for the while-read loop. The heredoc `while IFS= read -r ... done <<EOF` pattern is the POSIX-safe way (consistent with Phase 22/23 patterns in the codebase).
