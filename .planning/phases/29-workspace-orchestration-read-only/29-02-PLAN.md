ws_do_check <manifest_path> <manifest_dir> function:
  Purpose: reads repos from manifest, runs conjure check --porcelain per repo, emits table.
  Exit codes: 0 = all repos clean; 1 = any repo drifted or errored (partial-success, fail-tolerant); 2 = hard error (invalid manifest, etc.)
  NOTE: exit 1 here is the SC-MANDATED documented exception — it is the "partial success" aggregate semantic, NOT a hard failure. This mirrors how audit's WARN→exit-1 gate is the project's documented exception.

  Logic:
    OVERALL_RC=0
    printf '\n%-30s %-15s %s\n' "REPO" "STATUS" "EXIT"
    printf '%-30s %-15s %s\n' "----" "------" "----"

    Iterate repos in manifest order using jq:
    while IFS= read -r repo_json; do
      repo_name=$(printf '%s' "$repo_json" | jq -r '.name')
      repo_relpath=$(printf '%s' "$repo_json" | jq -r '.path')
      repo_abs="$MANIFEST_DIR/$repo_relpath"

      BAD-PATH GUARD: if [ ! -d "$repo_abs" ]; then
        printf '%-30s %-15s %s\n' "$repo_name" "SKIP" "bad-path"
        printf '  ⚠ skipping %s: path not found (%s)\n' "$repo_name" "$repo_abs" >&2
        OVERALL_RC=1  # partial success
        continue
      fi

      Per-repo check invocation (CONJURE_HOME propagated):
        REPO_RC=0
        PORCELAIN_OUT=$(CONJURE_HOME="$CONJURE_HOME" CONJURE_PORCELAIN=1 \
          bash "$CONJURE_HOME/cli/conjure" check "$repo_abs" 2>/dev/null) || REPO_RC=$?

      Map exit code to status label:
        case "$REPO_RC" in
          0) REPO_STATUS="clean" ;;
          1) REPO_STATUS="drift"; OVERALL_RC=1 ;;
          2) REPO_STATUS="error"; OVERALL_RC=1 ;;  # NOTE: per-repo exit 2 maps to overall exit 1 (fail-tolerant)
          *) REPO_STATUS="error($REPO_RC)"; OVERALL_RC=1 ;;
        esac

      Emit table row:
        printf '%-30s %-15s %s\n' "$repo_name" "$REPO_STATUS" "$REPO_RC"

    done < <(jq -c '.repos[]' "$MANIFEST_PATH")

    printf '\n'
    if [ "$OVERALL_RC" -eq 0 ]; then
      echo "✓ All repos clean."
    else
      echo "⚠ One or more repos have drift or errors (exit 1 — partial success)."
    fi

    return "$OVERALL_RC"

  audit)
    MANIFEST_PATH=""
    FAIL_FAST=0
    while [ $# -gt 0 ]; do
      case "$1" in
        --fail-fast) FAIL_FAST=1 ;;
        --help|-h)   echo "Usage: conjure workspace audit [--fail-fast] <manifest_path>"; exit 0 ;;
        *)           MANIFEST_PATH="$1" ;;
      esac; shift
    done
    [ -z "$MANIFEST_PATH" ] && MANIFEST_PATH="$(pwd)"
    MANIFEST_PATH="$(workspace_manifest_load "$MANIFEST_PATH")" || exit 2
    MANIFEST_DIR="$(dirname "$MANIFEST_PATH")"
    ws_do_audit "$MANIFEST_PATH" "$MANIFEST_DIR" "$FAIL_FAST"
    ;;

ws_do_audit <manifest_path> <manifest_dir> <fail_fast> function:
  Purpose: reads repos from manifest, runs conjure audit --json per repo, emits pass/fail table + global summary.
  Exit codes: 0 = all pass; 1 = at least one warn but no fail; 2 = at least one fail (or hard error).
  NOTE: exit 1 here is the SC-MANDATED documented exception for warn-only aggregate state.

  Logic:
    OVERALL_RC=0
    FAIL_COUNT=0
    WARN_COUNT=0
    PASS_COUNT=0
    SKIP_COUNT=0

    printf '\n%-30s %-15s %s\n' "REPO" "STATUS" "EXIT"
    printf '%-30s %-15s %s\n' "----" "------" "----"

    TMPJSON="$(mktemp)"
    _ws_audit_cleanup() { rm -f "${TMPJSON:-}"; }
    trap _ws_audit_cleanup EXIT

    while IFS= read -r repo_json; do
      repo_name=$(printf '%s' "$repo_json" | jq -r '.name')
      repo_relpath=$(printf '%s' "$repo_json" | jq -r '.path')
      repo_abs="$MANIFEST_DIR/$repo_relpath"

      BAD-PATH GUARD: if [ ! -d "$repo_abs" ]; then
        printf '%-30s %-15s %s\n' "$repo_name" "SKIP" "bad-path"
        printf '  ⚠ skipping %s: path not found (%s)\n' "$repo_name" "$repo_abs" >&2
        SKIP_COUNT=$((SKIP_COUNT+1))
        continue
      fi

      Per-repo audit --json invocation:
        REPO_RC=0
        CONJURE_HOME="$CONJURE_HOME" CONJURE_JSON=1 \
          bash "$CONJURE_HOME/cli/conjure" audit "$repo_abs" \
          >"$TMPJSON" 2>/dev/null || REPO_RC=$?

        Parse .status from captured JSON (TMPJSON):
        REPO_STATUS="unknown"
        if jq empty "$TMPJSON" >/dev/null 2>&1; then
          REPO_STATUS="$(jq -r '.status // "unknown"' "$TMPJSON" 2>/dev/null || echo "unknown")"
        fi

        Map to table display and counters:
        case "$REPO_STATUS" in
          pass)    PASS_COUNT=$((PASS_COUNT+1)); TABLE_STATUS="pass" ;;
          warn)    WARN_COUNT=$((WARN_COUNT+1)); TABLE_STATUS="warn"; [ "$OVERALL_RC" -lt 1 ] && OVERALL_RC=1 ;;
          fail)    FAIL_COUNT=$((FAIL_COUNT+1)); TABLE_STATUS="FAIL"; OVERALL_RC=2 ;;
          *)       TABLE_STATUS="error($REPO_RC)"; FAIL_COUNT=$((FAIL_COUNT+1)); OVERALL_RC=2 ;;
        esac

        printf '%-30s %-15s %s\n' "$repo_name" "$TABLE_STATUS" "$REPO_RC"

        --fail-fast: if [ "$FAIL_FAST" -eq 1 ] && [ "$OVERALL_RC" -eq 2 ]; then
          printf '\n✗ --fail-fast: aborting after first failure (%s)\n' "$repo_name" >&2
          return 2
        fi

    done < <(jq -c '.repos[]' "$MANIFEST_PATH")

    trap - EXIT; rm -f "$TMPJSON"

    GLOBAL SUMMARY:
    printf '\n── Workspace Audit Summary ──\n'
    printf '  Pass:  %d\n' "$PASS_COUNT"
    printf '  Warn:  %d\n' "$WARN_COUNT"
    printf '  Fail:  %d\n' "$FAIL_COUNT"
    printf '  Skip:  %d\n' "$SKIP_COUNT"
    if [ "$OVERALL_RC" -eq 0 ]; then echo "✓ All repos passed."
    elif [ "$OVERALL_RC" -eq 1 ]; then echo "⚠ Warnings only (exit 1 — partial success)."
    else echo "✗ One or more repos FAILED audit (exit 2)."; fi

    return "$OVERALL_RC"
